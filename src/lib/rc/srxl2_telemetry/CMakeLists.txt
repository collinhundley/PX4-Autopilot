# Also usable as a standalone host test project, without PX4 or uORB.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	cmake_minimum_required(VERSION 3.16)
	project(srxl2_telemetry_tests LANGUAGES CXX)
	enable_testing()
	set(SRXL2_TELEMETRY_STANDALONE ON)
endif()

add_library(srxl2_telemetry Telemetry.cpp)
target_include_directories(srxl2_telemetry PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(srxl2_telemetry PUBLIC cxx_std_14)
if(TARGET prebuild_targets)
	target_link_libraries(srxl2_telemetry PRIVATE prebuild_targets)
endif()

if(SRXL2_TELEMETRY_STANDALONE OR (BUILD_TESTING AND PX4_PLATFORM STREQUAL "posix"))
	option(SRXL2_TELEMETRY_SANITIZE "Run host tests with address and undefined-behavior sanitizers" OFF)
	target_compile_options(srxl2_telemetry PRIVATE -Wall -Wextra -Werror)
	add_executable(srxl2_telemetry_test TelemetryTest.cpp)
	target_link_libraries(srxl2_telemetry_test PRIVATE srxl2_telemetry)
	target_compile_options(srxl2_telemetry_test PRIVATE -Wall -Wextra -Werror)
	if(SRXL2_TELEMETRY_SANITIZE)
		target_compile_options(srxl2_telemetry PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
		target_link_options(srxl2_telemetry PUBLIC -fsanitize=address,undefined)
	endif()
	add_test(NAME srxl2_telemetry_test COMMAND srxl2_telemetry_test)
	if(TARGET test_results)
		if(COMMAND add_filtered_test_dependencies)
			add_filtered_test_dependencies(srxl2_telemetry_test)
		elseif(NOT TESTFILTER OR "srxl2_telemetry_test" MATCHES "${TESTFILTER}")
			add_dependencies(test_results srxl2_telemetry_test)
		endif()
	endif()
endif()
